#include <SPI.h>
#include <MFRC522.h>
#include <ESP8266WiFi.h>
#include <ESP8266HTTPClient.h>
#include <ArduinoJson.h>

// Pin definitions for MFRC522
#define RST_PIN D3 // Reset pin for MFRC522
#define SS_PIN D4  // Slave Select pin for MFRC522
// Relay pin
#define RELAY_PIN D7 // Relay connected to GPIO D7

// WiFi credentials
const char* ssid = "YOUR_WIFI_SSID"; // Replace with your WiFi SSID
const char* password = "YOUR_WIFI_PASSWORD"; // Replace with your WiFi password

// Base URL for HTTP request
const char* baseUrl = "http://iot.arduino.org.tw:8888/bigdata/rfid/rfidpass.php?KEY=";

// Initialize MFRC522
MFRC522 mfrc522(SS_PIN, RST_PIN);

// Function 1: Read RFID card number
String readRFIDCard() {
  // Check if a card is present
  if (!mfrc522.PICC_IsNewCardPresent()) {
    return "";
  }
  // Select the card
  if (!mfrc522.PICC_ReadCardSerial()) {
    return "";
  }
  // Read card UID
  String cardNumber = "";
  for (byte i = 0; i < mfrc522.uid.size; i++) {
    // Convert each byte to two-digit hex and append
    if (mfrc522.uid.uidByte[i] < 0x10) {
      cardNumber += "0";
    }
    cardNumber += String(mfrc522.uid.uidByte[i], HEX);
  }
  cardNumber.toUpperCase();
  // Halt PICC
  mfrc522.PICC_HaltA();
  // Stop encryption on PCD
  mfrc522.PCD_StopAuthState();
  return cardNumber;
}

// Function 2: Construct HTTP GET URL with RFID card number
String constructUrl(String cardNumber) {
  if (cardNumber == "") {
    return "";
  }
  return String(baseUrl) + cardNumber;
}

// Function 3: Send HTTP GET request and get response
String sendHttpGetRequest(String url) 
{
  if (url == "") {
    return "";
  }
  WiFiClient client;
  HTTPClient http;
  http.begin(client, url);
  int httpCode = http.GET();
  String payload = "";
  if (httpCode == HTTP_CODE_OK) {
    payload = http.getString();
  }
  http.end();
  return payload;
}

// Function 4: Parse JSON response and control relay
bool processJsonResponse(String jsonResponse, String cardNumber) {
  if (jsonResponse == "") {
    return false;
  }
  // Allocate the JSON document
  StaticJsonDocument<200> doc;
  // Parse JSON
  DeserializationError error = deserializeJson(doc, jsonResponse);
  if (error) {
    Serial.print("JSON parsing failed: ");
    Serial.println(error.c_str());
    return false;
  }
  // Extract Device and Result
  const char* device = doc["Device"];
  const char* result = doc["Result"];
  // Compare Device with cardNumber and check if Result is "Find"
  if (String(device) == cardNumber && String(result) == "Find") {
    digitalWrite(RELAY_PIN, HIGH); // Turn on relay
    return true;
  } else {
    digitalWrite(RELAY_PIN, LOW); // Ensure relay is off
    return false;
  }
}

void setup() {
  // Initialize Serial
  Serial.begin(115200);
  // Initialize SPI bus
  SPI.begin();
  // Initialize MFRC522
  mfrc522.PCD_Init();
  // Initialize relay pin
  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, LOW); // Ensure relay is off initially
  // Connect to WiFi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");
  while (WiFi.status() != WL_CONNECTED) {
    delay(500);
    Serial.print(".");
  }
  Serial.println("\nWiFi connected");
}

void loop() {
  // Function 1: Read RFID card
  String cardNumber = readRFIDCard();
  if (cardNumber != "") {
    Serial.println("Card Number: " + cardNumber);
    // Function 2: Construct URL
    String url = constructUrl(cardNumber);
    if (url != "") {
      Serial.println("URL: " + url);
      // Function 3: Send HTTP request
      String response = sendHttpGetRequest(url);
      if (response != "") {
        Serial.println("Response: " + response);
        // Function 4: Process JSON and control relay
        bool success = processJsonResponse(response, cardNumber);
        if (success) {
          Serial.println("Access granted: Relay ON");
        } else {
          Serial.println("Access denied: Relay OFF");
        }
      } else {
        Serial.println("HTTP request failed");
      }
    } else {
      Serial.println("Invalid URL");
    }
  }
  delay(1000); // Wait before next scan
}
